name: Build Windows app

on:
  workflow_dispatch:
  push:
    tags: ['v*']

jobs:
  build:
    runs-on: windows-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Cai thu vien
        run: npm install
      - name: Build file .exe
        run: npm run dist
      - name: Luu file cai dat
        uses: actions/upload-artifact@v4
        with:
          name: kho-de-vat-ly-windows
          path: dist/*.exe
      - name: Dang len Releases (khi tao tag v*)
        if: startsWith(github.ref, 'refs/tags/')
        uses: softprops/action-gh-release@v2
        with:
          files: dist/*.exe
